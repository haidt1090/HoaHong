## Screening Logic (Simplified)

```python
def screen_examinations(test_category, abnormal_only=True, **filters):
    exams = Examination.objects.filter(**filters).prefetch_related('department_results')
    results = []
    for exam in exams:
        abnormal_fields = []
        for dept_result in exam.department_results.all():
            template = dept_result.template_snapshot
            for field_def in template:
                if test_category and field_def.get('test_category') != test_category:
                    continue
                value = dept_result.data.get(field_def['key'])
                norm = field_def.get('normal_range')
                if norm and is_abnormal(value, norm):
                    abnormal_fields.append({
                        'department': dept_result.department.name,
                        'field_key': field_def['key'],
                        'field_label': field_def['label'],
                        'value': value,
                        'normal_range': norm,
                        'unit': field_def.get('unit', ''),
                    })
        if abnormal_fields:
            results.append({'examination': exam, 'abnormal_fields': abnormal_fields})
    return results
```
